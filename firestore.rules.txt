rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isLoggedIn() {
      return request.auth != null;
    }

    function userDoc() {
      return get(
        /databases/$(database)/documents/users/$(request.auth.uid)
      );
    }

    function hasUserProfile() {
      return isLoggedIn()
        && exists(
          /databases/$(database)/documents/users/$(request.auth.uid)
        );
    }

    function isActiveUser() {
      return hasUserProfile()
        && userDoc().data.active == true;
    }

    function isAdmin() {
      return isActiveUser()
        && userDoc().data.role == "admin";
    }

    function isBranchUser() {
      return isActiveUser()
        && userDoc().data.role == "branch";
    }

    function currentBranchId() {
      return userDoc().data.branchId;
    }

    function currentBranchDoc() {
      return get(
        /databases/$(database)/documents/branches/$(currentBranchId())
      );
    }

    function currentServiceGroup() {
      return currentBranchDoc().data.serviceGroup;
    }


    match /users/{userId} {
      allow read: if isLoggedIn()
        && (
          request.auth.uid == userId
          || isAdmin()
        );

      allow create, update, delete:
        if isAdmin();
    }


    match /branches/{branchId} {
      allow read:
        if isAdmin()
        || (
          isBranchUser()
          && branchId == currentBranchId()
        );

      allow create, update, delete:
        if isAdmin();
    }


    match /barbers/{barberId} {
      allow read:
        if isAdmin()
        || isBranchUser();

      allow create, update, delete:
        if isAdmin();
    }


    match /services/{serviceId} {
      allow read:
        if isAdmin()
        || (
          isBranchUser()
          && resource.data.groupId
            == currentServiceGroup()
        );

      allow create, update, delete:
        if isAdmin();
    }


    match /barber_presence/{barberId} {
      allow read:
        if isAdmin()
        || isBranchUser();

      allow create, update:
        if isAdmin()
        || (
          isBranchUser()
          && request.resource.data.barberId == barberId
          && request.resource.data.branchId == currentBranchId()
        );

      allow delete:
        if isAdmin();
    }


    match /daily_closings/{closingId} {
      allow read:
        if isAdmin()
        || (
          isBranchUser()
          && resource.data.branchId == currentBranchId()
        );

      allow create, update:
        if isAdmin()
        || (
          isBranchUser()
          && request.resource.data.branchId == currentBranchId()
        );

      allow delete:
        if isAdmin();
    }


    match /transactions/{transactionId} {
      allow read, write: if false;
    }


    match /{document=**} {
      allow read, write: if false;
    }
  }
}
